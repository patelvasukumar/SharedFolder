user_account () {

	local checker_user=0

	while test "$checker_user" -eq 0 ; do

	read -p "Enter the user name: " username

	if cat /etc/passwd | cut -d ":" -f 1 | fgrep -w "$username"  > /dev/null ; then

		echo ""
	    	echo "the user is already on the system. Please select another username."
        	echo ""

		sleep 3

	else 
		checker_user=1
	fi
	
	done

	read -p "Enter the home directory: (absolute path) " home_directory
	
	shell_loop=0

	while test "$shell_loop" -eq 0 ; do	
			
		read -p "Enter the default shell path: (absolute path) " shell
	
	if  cat /etc/shells | fgrep -x "$shell" > /dev/null ; then 

		shell_loop=1

	else
		echo ""
		echo " The shell is not registed in /etc/shells directory. Please, Try again ! "
		echo ""

	fi
	done

	sudo useradd -m -d "$home_directory" -s "$shell" "$username"
	
	if test "$?" -ne 0 ; then
		error_report
		return
	fi


	echo ""
	echo "$username added successfully."
	echo ""
	
	sleep 3

	clear

}
