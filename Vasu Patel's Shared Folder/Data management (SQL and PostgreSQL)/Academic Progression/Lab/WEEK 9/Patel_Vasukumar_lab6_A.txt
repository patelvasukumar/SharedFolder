-- Orders Table 
create table Orders(
	id int primary key,
	order_date date
	);

-- Products Table 
create table Products(
	id int primary key,
	Name varchar(45) not null,
	Description varchar(45) not null,
	Price int
);

-- Order_line Table
create table Order_line(
	products_id int,
	orders_id int,
	quantity int ,
	price decimal,
	line_total decimal,
	foreign key(products_id) references products(id),
	foreign key(orders_id) references orders(id)
);
